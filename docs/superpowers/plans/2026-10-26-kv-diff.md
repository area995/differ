# KV Diff Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** SPA statica che confronta semanticamente due file JSON/YAML con opzioni selezionabili (chiavi, valori, whitespace, ordine chiavi, ordine array, tipi laschi).

**Architecture:** Core puro in `src/core/` (parse → modello `Node` con posizioni riga/colonna → `diff()` → `DiffEntry[]`), testato con Vitest. UI vanilla TS in `src/ui/` che costruisce DOM, ricalcola con debounce e rende vista side-by-side + lista differenze. Un solo parser (`yaml` di eemeli) per entrambi i formati: JSON è sottoinsieme di YAML 1.2.

**Tech Stack:** Vite, TypeScript (strict), Vitest, jsdom (solo test UI), `yaml` v2.

**Spec:** `docs/superpowers/specs/2026-10-26-kv-diff-design.md`

**Comandi:** tutti da eseguire nella root `projectx/`. Shell: Git Bash o PowerShell, i comandi `npm`/`npx`/`git` sono identici.

```ts
export type Format = 'json' | 'yaml';

/** 1-based line and column. */
export type Pos = { line: number; col: number };
export type Range = { start: Pos; end: Pos };

export type ScalarValue = string | number | boolean | null;

export type Entry = { key: string; keyRange: Range; value: Node };

export type Node =
  | { kind: 'object'; range: Range; entries: Entry[] }
  | { kind: 'array'; range: Range; items: Node[] }
  | { kind: 'scalar'; range: Range; value: ScalarValue };

export type ObjectNode = Extract<Node, { kind: 'object' }>;
export type ArrayNode = Extract<Node, { kind: 'array' }>;

export type PathSeg = string | number | { field: string; value: string };
export type Path = PathSeg[];

export type DiffKind = 'added' | 'removed' | 'changed' | 'type' | 'moved' | 'whitespace';

export type Side = { value?: unknown; range: Range };

export type DiffEntry = {
  kind: DiffKind;
  path: Path;
  /** Missing for 'added'. */
  a?: Side;
  /** Missing for 'removed'. */
  b?: Side;
  detail?: string;
};

export type DiffOptions = {
  keys: boolean;
  values: boolean;
  whitespace: boolean;
  keyOrder: boolean;
  ignoreArrayOrder: boolean;
  arrayKeyField: string;
  looseTypes: boolean;
};

export const DEFAULT_OPTIONS: DiffOptions = {
  keys: true,
  values: true,
  whitespace: false,
  keyOrder: false,
  ignoreArrayOrder: false,
  arrayKeyField: '',
  looseTypes: false,
};

export type ParseError = { message: string; pos: Pos };
export type ParseResult = { ok: true; node: Node } | { ok: false; error: ParseError };

const IDENT = /^[A-Za-z_$][A-Za-z0-9_$-]*$/;

export function formatPath(path: Path): string {
  let out = '$';
  for (const seg of path) {
    if (typeof seg === 'number') out += `[${seg}]`;
    else if (typeof seg === 'string') out += IDENT.test(seg) ? `.${seg}` : `[${JSON.stringify(seg)}]`;
    else out += `[${seg.field}=${seg.value}]`;
  }
  return out;
}

export function toPlain(node: Node): unknown {
  switch (node.kind) {
    case 'scalar':
      return node.value;
    case 'array':
      return node.items.map(toPlain);
    case 'object':
      return Object.fromEntries(node.entries.map((e) => [e.key, toPlain(e.value)]));
  }
}
```

```ts
import type { Format } from './model';

export function detectFormat(text: string, fileName?: string): Format {
  const ext = fileName?.toLowerCase().split('.').pop();
  if (ext === 'json') return 'json';
  if (ext === 'yaml' || ext === 'yml') return 'yaml';
  const trimmed = text.trim();
  if (trimmed.startsWith('{') || trimmed.startsWith('[')) {
    try {
      JSON.parse(trimmed);
      return 'json';
    } catch {
      // not JSON: flow-style YAML
    }
  }
  return 'yaml';
}
```

```ts
import {
  LineCounter,
  isAlias,
  isMap,
  isNode,
  isScalar,
  isSeq,
  parseAllDocuments,
  type Document,
} from 'yaml';
import type { Format, Node, ParseResult, Pos, Range, ScalarValue } from './model';

type Ctx = { doc: Document.Parsed; lc: LineCounter; text: string; stack: Set<unknown> };

const START: Pos = { line: 1, col: 1 };

export function parse(text: string, format: Format): ParseResult {
  if (format === 'json') {
    try {
      JSON.parse(text);
    } catch (e) {
      const message = (e as Error).message;
      return { ok: false, error: { message, pos: jsonErrorPos(message, text) } };
    }
  }
  const lc = new LineCounter();
  const docs = parseAllDocuments(text, { lineCounter: lc, uniqueKeys: true, prettyErrors: false });
  if (docs.length === 0) return { ok: true, node: { kind: 'scalar', value: null, range: point(START) } };
  if (docs.length > 1) {
    return {
      ok: false,
      error: { message: 'YAML multi-documento non supportato', pos: lc.linePos(docs[1].range[0]) },
    };
  }
  const doc = docs[0];
  const err = doc.errors[0];
  if (err) return { ok: false, error: { message: err.message, pos: lc.linePos(err.pos[0]) } };
  try {
    return { ok: true, node: convert(doc.contents, { doc, lc, text, stack: new Set() }) };
  } catch (e) {
    return { ok: false, error: { message: (e as Error).message, pos: START } };
  }
}

function convert(n: unknown, ctx: Ctx): Node {
  if (n == null) return { kind: 'scalar', value: null, range: point(START) };
  if (isAlias(n)) {
    const target = n.resolve(ctx.doc);
    if (!target) throw new Error(`Alias non risolto: *${n.source}`);
    if (ctx.stack.has(target)) throw new Error(`Alias ciclico: *${n.source}`);
    return { ...convert(target, ctx), range: toRange(n.range, ctx) };
  }
  if (isMap(n)) {
    ctx.stack.add(n);
    const entries = n.items.map((pair) => {
      const k = pair.key;
      const key = isScalar(k)
        ? String(k.value)
        : k == null
          ? ''
          : JSON.stringify((k as { toJSON(): unknown }).toJSON());
      const keyRange = toRange(isNode(k) ? k.range : undefined, ctx);
      const value: Node =
        pair.value == null
          ? { kind: 'scalar', value: null, range: point(keyRange.end) }
          : convert(pair.value, ctx);
      return { key, keyRange, value };
    });
    ctx.stack.delete(n);
    return { kind: 'object', range: toRange(n.range, ctx), entries };
  }
  if (isSeq(n)) {
    ctx.stack.add(n);
    const items = n.items.map((item) => convert(item, ctx));
    ctx.stack.delete(n);
    return { kind: 'array', range: toRange(n.range, ctx), items };
  }
  if (isScalar(n)) return { kind: 'scalar', value: toScalarValue(n.value), range: toRange(n.range, ctx) };
  return { kind: 'scalar', value: String(n), range: point(START) };
}

function toScalarValue(v: unknown): ScalarValue {
  if (v === undefined) return null;
  if (v === null || typeof v === 'string' || typeof v === 'number' || typeof v === 'boolean') return v;
  if (typeof v === 'bigint') return Number(v);
  return String(v);
}

function toRange(r: readonly [number, number, number] | null | undefined, ctx: Ctx): Range {
  if (!r) return point(START);
  let end = r[1];
  while (end > r[0] && /\s/.test(ctx.text[end - 1])) end--;
  return { start: ctx.lc.linePos(r[0]), end: ctx.lc.linePos(end) };
}

function point(p: Pos): Range {
  return { start: p, end: p };
}

/** V8 messages look like "... in JSON at position 12 (line 3 column 1)". */
function jsonErrorPos(message: string, text: string): Pos {
  const m = /position (\d+)/.exec(message);
  if (!m) return START;
  const offset = Number(m[1]);
  const before = text.slice(0, offset);
  return { line: before.split('\n').length, col: offset - before.lastIndexOf('\n') };
}
```

```ts
import type { Node, ScalarValue } from './model';

export type ScalarType = 'string' | 'number' | 'boolean' | 'null';

export function scalarType(v: ScalarValue): ScalarType {
  return v === null ? 'null' : (typeof v as Exclude<ScalarType, 'null'>);
}

export function scalarEqual(x: ScalarValue, y: ScalarValue, loose: boolean): boolean {
  if (x === y) return true;
  return loose && normalize(x) === normalize(y);
}

function normalize(v: ScalarValue): string {
  if (typeof v !== 'string') return String(v);
  const t = v.trim();
  if (t === '') return '';
  const lower = t.toLowerCase();
  if (lower === 'true' || lower === 'false' || lower === 'null') return lower;
  const n = Number(t);
  return Number.isNaN(n) ? v : String(n);
}

export function nodeEqual(a: Node, b: Node, loose: boolean): boolean {
  if (a.kind === 'scalar' && b.kind === 'scalar') return scalarEqual(a.value, b.value, loose);
  if (a.kind === 'array' && b.kind === 'array') {
    return a.items.length === b.items.length && a.items.every((x, i) => nodeEqual(x, b.items[i], loose));
  }
  if (a.kind === 'object' && b.kind === 'object') {
    if (a.entries.length !== b.entries.length) return false;
    const valuesB = new Map(b.entries.map((e) => [e.key, e.value]));
    return a.entries.every((e) => {
      const v = valuesB.get(e.key);
      return v !== undefined && nodeEqual(e.value, v, loose);
    });
  }
  return false;
}
```

```ts
/**
 * Given the same set of keys in A order and B order, returns the keys (in A order)
 * that are not part of the longest common subsequence, i.e. the ones that moved.
 */
export function movedKeys(a: string[], b: string[]): string[] {
  const n = a.length;
  const m = b.length;
  const dp = Array.from({ length: n + 1 }, () => new Array<number>(m + 1).fill(0));
  for (let i = n - 1; i >= 0; i--) {
    for (let j = m - 1; j >= 0; j--) {
      dp[i][j] = a[i] === b[j] ? dp[i + 1][j + 1] + 1 : Math.max(dp[i + 1][j], dp[i][j + 1]);
    }
  }
  const stable = new Set<string>();
  let i = 0;
  let j = 0;
  while (i < n && j < m) {
    if (a[i] === b[j]) {
      stable.add(a[i]);
      i++;
      j++;
    } else if (dp[i + 1][j] >= dp[i][j + 1]) i++;
    else j++;
  }
  return a.filter((k) => !stable.has(k));
}
```

```ts
import { nodeEqual } from './equal';
import type { DiffOptions, Node, PathSeg } from './model';

export type ItemRef = { index: number; seg: PathSeg };
export type ItemPair = { a: number; b: number; seg: PathSeg };
export type ArrayMatch = { pairs: ItemPair[]; onlyA: ItemRef[]; onlyB: ItemRef[] };

type MatchOptions = Pick<DiffOptions, 'ignoreArrayOrder' | 'arrayKeyField' | 'looseTypes'>;

export function matchArrays(a: Node[], b: Node[], opts: MatchOptions): ArrayMatch {
  const res: ArrayMatch = { pairs: [], onlyA: [], onlyB: [] };
  if (!opts.ignoreArrayOrder) {
    zip(indexes(a.length), indexes(b.length), res);
    return res;
  }

  let restA = indexes(a.length);
  let restB = indexes(b.length);

  const field = opts.arrayKeyField.trim();
  if (field) {
    const byField = new Map<string, number>();
    for (const j of restB) {
      const v = fieldValue(b[j], field);
      if (v !== undefined && !byField.has(v)) byField.set(v, j);
    }
    const usedByField = new Set<number>();
    const leftA: number[] = [];
    for (const i of restA) {
      const v = fieldValue(a[i], field);
      if (v === undefined) {
        leftA.push(i);
        continue;
      }
      const seg = { field, value: v };
      const j = byField.get(v);
      if (j !== undefined && !usedByField.has(j)) {
        usedByField.add(j);
        res.pairs.push({ a: i, b: j, seg });
      } else res.onlyA.push({ index: i, seg });
    }
    const leftB: number[] = [];
    for (const j of restB) {
      if (usedByField.has(j)) continue;
      const v = fieldValue(b[j], field);
      if (v === undefined) leftB.push(j);
      else res.onlyB.push({ index: j, seg: { field, value: v } });
    }
    restA = leftA;
    restB = leftB;
  }

  const usedByEquality = new Set<number>();
  const leftA: number[] = [];
  for (const i of restA) {
    const j = restB.find((k) => !usedByEquality.has(k) && nodeEqual(a[i], b[k], opts.looseTypes));
    if (j === undefined) leftA.push(i);
    else {
      usedByEquality.add(j);
      res.pairs.push({ a: i, b: j, seg: i });
    }
  }
  zip(
    leftA,
    restB.filter((k) => !usedByEquality.has(k)),
    res,
  );
  res.pairs.sort((x, y) => x.a - y.a);
  return res;
}

function fieldValue(node: Node, field: string): string | undefined {
  if (node.kind !== 'object') return undefined;
  const entry = node.entries.find((e) => e.key === field);
  return entry && entry.value.kind === 'scalar' ? String(entry.value.value) : undefined;
}

function indexes(n: number): number[] {
  return Array.from({ length: n }, (_, i) => i);
}

function zip(as: number[], bs: number[], res: ArrayMatch): void {
  const n = Math.min(as.length, bs.length);
  for (let k = 0; k < n; k++) res.pairs.push({ a: as[k], b: bs[k], seg: as[k] });
  for (const i of as.slice(n)) res.onlyA.push({ index: i, seg: i });
  for (const j of bs.slice(n)) res.onlyB.push({ index: j, seg: j });
}
```

```ts
/** Describes the whitespace-only differences between two lines (empty if none). */
export function compareLines(a: string, b: string): string[] {
  if (a === b) return [];
  const details: string[] = [];

  const indentA = leading(a);
  const indentB = leading(b);
  if (indentA.includes('\t') !== indentB.includes('\t')) details.push("tab vs spazi nell'indentazione");
  else if (indentA !== indentB) details.push(`indentazione ${indentA.length} → ${indentB.length}`);

  const trailA = trailing(a, indentA.length);
  const trailB = trailing(b, indentB.length);
  if (trailA !== trailB) details.push('spazi finali');

  const coreA = a.slice(indentA.length, a.length - trailA.length);
  const coreB = b.slice(indentB.length, b.length - trailB.length);
  if (coreA !== coreB && coreA.replace(/\s+/g, '') === coreB.replace(/\s+/g, '')) {
    details.push('spaziatura interna');
  }
  return details;
}

function leading(s: string): string {
  return /^[ \t]*/.exec(s)![0];
}

function trailing(s: string, from: number): string {
  return /[ \t]*$/.exec(s.slice(from))![0];
}
```

```ts
import { matchArrays } from './arrays';
import { scalarEqual, scalarType } from './equal';
import {
  toPlain,
  type ArrayNode,
  type DiffEntry,
  type DiffOptions,
  type Entry,
  type Node,
  type ObjectNode,
  type Path,
  type Range,
  type Side,
} from './model';
import { movedKeys } from './order';
import { compareLines } from './whitespace';

/** Original texts, needed only for whitespace comparison. */
export type RawTexts = { a: string; b: string };

type Ctx = {
  opts: DiffOptions;
  out: DiffEntry[];
  ws?: { linesA: string[]; linesB: string[]; seen: Set<string> };
};

export function diff(a: Node, b: Node, opts: DiffOptions, raw?: RawTexts): DiffEntry[] {
  const ctx: Ctx = { opts, out: [] };
  if (opts.whitespace && raw) {
    ctx.ws = { linesA: raw.a.split(/\r?\n/), linesB: raw.b.split(/\r?\n/), seen: new Set() };
  }
  walk(a, b, [], ctx);
  return ctx.out;
}

function walk(a: Node, b: Node, path: Path, ctx: Ctx): void {
  const { opts, out } = ctx;
  if (a.kind === 'scalar' && b.kind === 'scalar') {
    if (scalarEqual(a.value, b.value, opts.looseTypes)) {
      checkWhitespace(a.range.start.line, b.range.start.line, path, ctx);
      return;
    }
    if (!opts.values) return;
    const ta = scalarType(a.value);
    const tb = scalarType(b.value);
    const sameType = ta === tb || opts.looseTypes;
    out.push({
      kind: sameType ? 'changed' : 'type',
      path,
      a: side(a),
      b: side(b),
      detail: sameType ? undefined : `${ta} → ${tb}`,
    });
    return;
  }
  if (a.kind === 'object' && b.kind === 'object') return walkObject(a, b, path, ctx);
  if (a.kind === 'array' && b.kind === 'array') return walkArray(a, b, path, ctx);
  if (opts.values) {
    out.push({ kind: 'type', path, a: side(a), b: side(b), detail: `${kindLabel(a)} → ${kindLabel(b)}` });
  }
}

function walkObject(a: ObjectNode, b: ObjectNode, path: Path, ctx: Ctx): void {
  const { opts, out } = ctx;
  const indexB = new Map(b.entries.map((e, i) => [e.key, i]));
  const keysA = new Set(a.entries.map((e) => e.key));

  for (const ea of a.entries) {
    const p = [...path, ea.key];
    const ib = indexB.get(ea.key);
    if (ib === undefined) {
      if (opts.keys) out.push({ kind: 'removed', path: p, a: entrySide(ea) });
      continue;
    }
    const eb = b.entries[ib];
    if (ea.value.kind !== 'scalar' && eb.value.kind !== 'scalar') {
      checkWhitespace(ea.keyRange.start.line, eb.keyRange.start.line, p, ctx);
    }
    walk(ea.value, eb.value, p, ctx);
  }

  if (opts.keys) {
    for (const eb of b.entries) {
      if (!keysA.has(eb.key)) out.push({ kind: 'added', path: [...path, eb.key], b: entrySide(eb) });
    }
  }

  if (opts.keyOrder) {
    const indexA = new Map(a.entries.map((e, i) => [e.key, i]));
    const commonA = a.entries.filter((e) => indexB.has(e.key)).map((e) => e.key);
    const commonB = b.entries.filter((e) => keysA.has(e.key)).map((e) => e.key);
    for (const key of movedKeys(commonA, commonB)) {
      const ia = indexA.get(key)!;
      const ib = indexB.get(key)!;
      out.push({
        kind: 'moved',
        path: [...path, key],
        a: { range: entryRange(a.entries[ia]) },
        b: { range: entryRange(b.entries[ib]) },
        detail: `posizione ${ia + 1} → ${ib + 1}`,
      });
    }
  }
}

function walkArray(a: ArrayNode, b: ArrayNode, path: Path, ctx: Ctx): void {
  const m = matchArrays(a.items, b.items, ctx.opts);
  for (const p of m.pairs) walk(a.items[p.a], b.items[p.b], [...path, p.seg], ctx);
  if (!ctx.opts.keys) return;
  for (const r of m.onlyA) ctx.out.push({ kind: 'removed', path: [...path, r.seg], a: side(a.items[r.index]) });
  for (const r of m.onlyB) ctx.out.push({ kind: 'added', path: [...path, r.seg], b: side(b.items[r.index]) });
}

function checkWhitespace(lineA: number, lineB: number, path: Path, ctx: Ctx): void {
  const ws = ctx.ws;
  if (!ws) return;
  const id = `${lineA}:${lineB}`;
  if (ws.seen.has(id)) return;
  ws.seen.add(id);
  const details = compareLines(ws.linesA[lineA - 1] ?? '', ws.linesB[lineB - 1] ?? '');
  if (details.length === 0) return;
  ctx.out.push({
    kind: 'whitespace',
    path,
    a: { range: lineRange(lineA) },
    b: { range: lineRange(lineB) },
    detail: details.join(', '),
  });
}

function side(n: Node): Side {
  return { value: toPlain(n), range: n.range };
}

function entrySide(e: Entry): Side {
  return { value: toPlain(e.value), range: entryRange(e) };
}

function entryRange(e: Entry): Range {
  return { start: e.keyRange.start, end: e.value.range.end };
}

function lineRange(line: number): Range {
  return { start: { line, col: 1 }, end: { line, col: 1 } };
}

function kindLabel(n: Node): string {
  return n.kind === 'scalar' ? scalarType(n.value) : n.kind;
}
```

```ts
import type { DiffOptions } from '../core/model';

type BoolKey = Exclude<keyof DiffOptions, 'arrayKeyField'>;

const CHECKS: { key: BoolKey; label: string; title: string }[] = [
  { key: 'keys', label: 'Differenze chiavi', title: 'Chiavi aggiunte o rimosse' },
  { key: 'values', label: 'Differenze valori', title: 'Valori o tipi diversi sulle stesse chiavi' },
  { key: 'whitespace', label: 'Spazi e tabulazioni', title: 'Differenze di formattazione su contenuti uguali' },
  { key: 'keyOrder', label: 'Posizione chiavi', title: 'Chiavi comuni in ordine diverso' },
  { key: 'ignoreArrayOrder', label: 'Ignora ordine array', title: "Abbina gli elementi degli array senza considerare l'indice" },
  { key: 'looseTypes', label: 'Tipi laschi', title: '8080 uguale a "8080", true uguale a "true"' },
];

export type OptionsBar = { setWhitespaceEnabled(enabled: boolean, reason: string): void };

export function createOptionsBar(
  root: HTMLElement,
  initial: DiffOptions,
  onChange: (opts: DiffOptions) => void,
): OptionsBar {
  const opts = { ...initial };
  const labels = new Map<BoolKey, HTMLLabelElement>();
  const inputs = new Map<BoolKey, HTMLInputElement>();

  const fieldInput = el('input');
  fieldInput.type = 'text';
  fieldInput.placeholder = 'campo chiave (es. name)';
  fieldInput.value = opts.arrayKeyField;
  fieldInput.disabled = !opts.ignoreArrayOrder;
  fieldInput.addEventListener('input', () => {
    opts.arrayKeyField = fieldInput.value;
    onChange({ ...opts });
  });

  for (const c of CHECKS) {
    const label = el('label', 'opt');
    label.title = c.title;
    const input = el('input');
    input.type = 'checkbox';
    input.checked = opts[c.key];
    input.addEventListener('change', () => {
      opts[c.key] = input.checked;
      fieldInput.disabled = !opts.ignoreArrayOrder;
      onChange({ ...opts });
    });
    label.append(input, c.label);
    root.append(label);
    labels.set(c.key, label);
    inputs.set(c.key, input);
  }

  const field = el('label', 'opt');
  field.title = 'Con "Ignora ordine array": abbina gli oggetti per questo campo (es. name, id)';
  field.append(fieldInput);
  root.append(field);

  return {
    setWhitespaceEnabled(enabled, reason) {
      const label = labels.get('whitespace')!;
      inputs.get('whitespace')!.disabled = !enabled;
      label.classList.toggle('disabled', !enabled);
      label.title = enabled ? CHECKS[2].title : reason;
    },
  };
}
```

```ts
import type { Format, ParseError } from '../core/model';
import { el } from './dom';

export type FormatChoice = 'auto' | Format;
export type InputState = { text: string; fileName?: string; choice: FormatChoice };
export type InputPanel = { setStatus(format: Format | null, error: ParseError | null): void };

const MAX_SIZE = 5 * 1024 * 1024;

export function createInputPanel(
  root: HTMLElement,
  title: string,
  onChange: (state: InputState) => void,
): InputPanel {
  const state: InputState = { text: '', choice: 'auto' };

  const heading = el('h2');
  heading.textContent = title;
  const select = el('select');
  for (const [value, label] of [
    ['auto', 'Auto'],
    ['json', 'JSON'],
    ['yaml', 'YAML'],
  ]) {
    const option = el('option');
    option.value = value;
    option.textContent = label;
    select.append(option);
  }
  const detected = el('span', 'detected');
  const fileInput = el('input');
  fileInput.type = 'file';
  fileInput.accept = '.json,.yaml,.yml';
  fileInput.hidden = true;
  const fileButton = el('label', 'file-btn');
  fileButton.append('Apri file…', fileInput);
  const head = el('div', 'panel-head');
  head.append(heading, select, detected, fileButton);

  const gutter = el('pre', 'gutter');
  const textarea = el('textarea');
  textarea.spellcheck = false;
  textarea.wrap = 'off';
  textarea.placeholder = 'Incolla qui il contenuto o trascina un file';
  const editor = el('div', 'editor');
  editor.append(gutter, textarea);

  const error = el('div', 'error');
  error.hidden = true;
  root.append(head, editor, error);

  let lineCount = -1;
  let errorLine = 0;
  const renderGutter = (force = false) => {
    const n = textarea.value.split('\n').length;
    if (n !== lineCount || force) {
      lineCount = n;
      gutter.replaceChildren(
        ...Array.from({ length: n }, (_, i) => {
          const span = el('span', i + 1 === errorLine ? 'err' : undefined);
          span.textContent = String(i + 1);
          return span;
        }),
      );
    }
    gutter.scrollTop = textarea.scrollTop;
  };

  const emit = () => onChange({ ...state });

  const load = async (file: File) => {
    const mb = (file.size / (1024 * 1024)).toFixed(1);
    if (file.size > MAX_SIZE && !confirm(`${file.name} pesa ${mb} MB. Caricarlo comunque?`)) return;
    state.text = await file.text();
    state.fileName = file.name;
    textarea.value = state.text;
    renderGutter();
    emit();
  };

  textarea.addEventListener('input', () => {
    state.text = textarea.value;
    renderGutter();
    emit();
  });
  textarea.addEventListener('scroll', () => {
    gutter.scrollTop = textarea.scrollTop;
  });
  select.addEventListener('change', () => {
    state.choice = select.value as FormatChoice;
    emit();
  });
  fileInput.addEventListener('change', () => {
    const file = fileInput.files?.[0];
    if (file) void load(file);
    fileInput.value = '';
  });
  root.addEventListener('dragover', (e) => {
    e.preventDefault();
    root.classList.add('drop');
  });
  root.addEventListener('dragleave', () => root.classList.remove('drop'));
  root.addEventListener('drop', (e) => {
    e.preventDefault();
    root.classList.remove('drop');
    const file = e.dataTransfer?.files[0];
    if (file) void load(file);
  });

  renderGutter();

  return {
    setStatus(format, err) {
      detected.textContent = format && state.choice === 'auto' ? `rilevato: ${format.toUpperCase()}` : '';
      error.hidden = !err;
      error.textContent = err ? `Riga ${err.pos.line}, colonna ${err.pos.col}: ${err.message}` : '';
      const nextErrorLine = err?.pos.line ?? 0;
      if (nextErrorLine !== errorLine) {
        errorLine = nextErrorLine;
        renderGutter(true);
      }
    },
  };
}
```

```ts
import { formatPath, type DiffEntry, type DiffKind } from '../core/model';
import { el } from './dom';

const COUNT_LABELS: Record<DiffKind, string> = {
  added: 'aggiunte',
  removed: 'rimosse',
  changed: 'modificate',
  type: 'tipo cambiato',
  moved: 'spostate',
  whitespace: 'formattazione',
};

const ITEM_LABELS: Record<DiffKind, string> = {
  added: 'aggiunta',
  removed: 'rimossa',
  changed: 'modificata',
  type: 'tipo',
  moved: 'spostata',
  whitespace: 'formattazione',
};

/** When several differences touch the same line, the later kind wins. */
const PRIORITY: DiffKind[] = ['whitespace', 'moved', 'added', 'removed', 'changed', 'type'];

export type ResultInput = { textA: string; textB: string; entries: DiffEntry[] };

export function renderResult(root: HTMLElement, input: ResultInput | null): void {
  root.replaceChildren();
  if (!input) return;
  if (input.entries.length === 0) {
    const p = el('p', 'equal');
    p.textContent = 'File equivalenti con le opzioni correnti.';
    root.append(p);
    return;
  }
  const paneA = pane(input.textA, lineMarks(input.entries, 'a'), 'A');
  const paneB = pane(input.textB, lineMarks(input.entries, 'b'), 'B');
  const panes = el('div', 'panes');
  panes.append(paneA, paneB);
  root.append(summary(input.entries), panes, list(input.entries, paneA, paneB));
}

export function describeEntry(e: DiffEntry): string {
  const va = show(e.a?.value);
  const vb = show(e.b?.value);
  switch (e.kind) {
    case 'added':
      return vb;
    case 'removed':
      return va;
    case 'changed':
      return `${va} → ${vb}`;
    case 'type':
      return `${va} → ${vb} (${e.detail})`;
    default:
      return e.detail ?? '';
  }
}

function summary(entries: DiffEntry[]): HTMLElement {
  const counts = new Map<DiffKind, number>();
  for (const e of entries) counts.set(e.kind, (counts.get(e.kind) ?? 0) + 1);
  const div = el('div', 'summary');
  for (const kind of Object.keys(COUNT_LABELS) as DiffKind[]) {
    const n = counts.get(kind);
    if (!n) continue;
    const chip = el('span', `chip d-${kind}`);
    chip.textContent = `${n} ${COUNT_LABELS[kind]}`;
    div.append(chip);
  }
  return div;
}

function lineMarks(entries: DiffEntry[], side: 'a' | 'b'): Map<number, DiffKind> {
  const marks = new Map<number, DiffKind>();
  for (const e of entries) {
    const s = e[side];
    if (!s) continue;
    for (let line = s.range.start.line; line <= s.range.end.line; line++) {
      const current = marks.get(line);
      if (!current || PRIORITY.indexOf(e.kind) > PRIORITY.indexOf(current)) marks.set(line, e.kind);
    }
  }
  return marks;
}

function pane(text: string, marks: Map<number, DiffKind>, side: string): HTMLElement {
  const div = el('div', 'pane');
  div.dataset.side = side;
  text.split(/\r?\n/).forEach((line, i) => {
    const n = i + 1;
    const kind = marks.get(n);
    const row = el('div', kind ? `line d-${kind}` : 'line');
    row.dataset.line = String(n);
    const num = el('span', 'num');
    num.textContent = String(n);
    const code = el('span', 'code');
    code.textContent = kind === 'whitespace' ? line.replace(/\t/g, '→').replace(/ /g, '·') : line;
    row.append(num, code);
    div.append(row);
  });
  return div;
}

function list(entries: DiffEntry[], paneA: HTMLElement, paneB: HTMLElement): HTMLElement {
  const ul = el('ul', 'diff-list');
  for (const e of entries) {
    const li = el('li', `d-${e.kind}`);
    const kind = el('span', 'kind');
    kind.textContent = ITEM_LABELS[e.kind];
    const path = el('code', 'path');
    path.textContent = formatPath(e.path);
    const desc = el('span', 'desc');
    desc.textContent = describeEntry(e);
    li.append(kind, path, desc);
    li.addEventListener('click', () => {
      if (e.a) reveal(paneA, e.a.range.start.line);
      if (e.b) reveal(paneB, e.b.range.start.line);
    });
    ul.append(li);
  }
  return ul;
}

function reveal(pane: HTMLElement, line: number): void {
  const row = pane.querySelector<HTMLElement>(`[data-line="${line}"]`);
  if (!row) return;
  pane.scrollTop = row.offsetTop - pane.clientHeight / 2;
  row.classList.remove('flash');
  void row.offsetWidth; // restart the animation
  row.classList.add('flash');
}

function show(v: unknown): string {
  if (v === undefined) return '';
  const s = JSON.stringify(v);
  return s.length > 80 ? `${s.slice(0, 77)}…` : s;
}
```

```ts
import { detectFormat } from '../core/detect';
import { diff } from '../core/diff';
import { DEFAULT_OPTIONS, type DiffOptions, type Format, type ParseResult } from '../core/model';
import { parse } from '../core/parse';
import { createInputPanel, type InputState } from './input';
import { createOptionsBar } from './options';
import { renderResult } from './result';

const byId = (id: string) => document.getElementById(id)!;

let opts: DiffOptions = { ...DEFAULT_OPTIONS };
let inputA: InputState = { text: '', choice: 'auto' };
let inputB: InputState = { text: '', choice: 'auto' };
let timer: number | undefined;

const optionsBar = createOptionsBar(byId('options'), opts, (o) => {
  opts = o;
  schedule();
});
const panelA = createInputPanel(byId('input-a'), 'File A', (s) => {
  inputA = s;
  schedule();
});
const panelB = createInputPanel(byId('input-b'), 'File B', (s) => {
  inputB = s;
  schedule();
});

function schedule(): void {
  window.clearTimeout(timer);
  timer = window.setTimeout(run, 300);
}

function analyze(s: InputState): { format: Format; result: ParseResult } | null {
  if (s.text.trim() === '') return null;
  const format = s.choice === 'auto' ? detectFormat(s.text, s.fileName) : s.choice;
  return { format, result: parse(s.text, format) };
}

function run(): void {
  const a = analyze(inputA);
  const b = analyze(inputB);
  panelA.setStatus(a?.format ?? null, a && !a.result.ok ? a.result.error : null);
  panelB.setStatus(b?.format ?? null, b && !b.result.ok ? b.result.error : null);

  const sameFormat = !a || !b || a.format === b.format;
  optionsBar.setWhitespaceEnabled(sameFormat, 'Disponibile solo con file dello stesso formato');

  if (!a || !b || !a.result.ok || !b.result.ok) {
    renderResult(byId('result'), null);
    return;
  }
  const raw = sameFormat ? { a: inputA.text, b: inputB.text } : undefined;
  const entries = diff(a.result.node, b.result.node, opts, raw);
  renderResult(byId('result'), { textA: inputA.text, textB: inputB.text, entries });
}
```

````markdown
# KV Diff

Confronto semantico tra file chiave-valore (JSON, YAML) nel browser. I file non lasciano mai il client.

- **Differenze chiavi**: chiavi aggiunte o rimosse
- **Differenze valori**: valori o tipi diversi sulle stesse chiavi
- **Spazi e tabulazioni**: differenze di formattazione su contenuti uguali (solo file dello stesso formato)
- **Posizione chiavi**: chiavi comuni in ordine diverso
- **Ignora ordine array**: abbina gli elementi per uguaglianza, o per un campo chiave (es. `name`)
- **Tipi laschi**: `8080` uguale a `"8080"`
````
